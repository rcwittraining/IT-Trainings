Schema: SHOW TABLES, DESCRIBE employees.

Select: SELECT name, salary FROM employees;

Filter: SELECT * FROM employees WHERE salary > 50000 ORDER BY salary DESC;

Aggregate: SELECT dept_id, COUNT(*) FROM employees GROUP BY dept_id;

Join: SELECT e.name, d.name FROM employees e JOIN departments d ON e.dept_id = d.id;
